cmake_minimum_required(VERSION 3.16)
project(WinTunes VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

set(_wintunes_optional_default ON)
if(WIN32)
    set(_wintunes_optional_default OFF)
endif()
option(WINTUNES_ENABLE_IPOD "Enable libgpod/iPod integration" ${_wintunes_optional_default})
option(WINTUNES_ENABLE_CD "Enable libcdio CD ripping/burning" ${_wintunes_optional_default})
option(WINTUNES_ENABLE_RADIO "Enable libVLC internet radio" ${_wintunes_optional_default})

find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK3 REQUIRED gtk+-3.0)
pkg_check_modules(GST REQUIRED gstreamer-1.0 gstreamer-audio-1.0)
pkg_check_modules(TAGLIB REQUIRED taglib)
pkg_check_modules(SQLITE3 REQUIRED sqlite3)

# These libraries are not available in every MSYS2 repository.  Features using
# them remain in the UI but return a clear unsupported error when absent.
pkg_check_modules(GPOD libgpod-1.0)
pkg_check_modules(CDIO libcdio)
pkg_check_modules(CDIO_PARA libcdio_paranoia)
pkg_check_modules(CDDB libcddb)
pkg_check_modules(VLC libvlc)

set(SOURCES src/main.cpp src/Database.cpp src/Library.cpp
    src/AudioPlayer.cpp src/RadioPlayer.cpp src/iPodSync.cpp
    src/iPodRestore.cpp src/CDManager.cpp src/MainWindow.cpp)
set(HEADERS src/Track.h src/Database.h src/Library.h src/AudioPlayer.h
    src/RadioPlayer.h src/iPodSync.h src/iPodRestore.h src/CDManager.h
    src/MainWindow.h)
add_executable(wintunes ${SOURCES} ${HEADERS})

set(PKG_MODULES GTK3 GST TAGLIB SQLITE3)
foreach(mod IN LISTS PKG_MODULES)
    target_compile_options(wintunes PRIVATE ${${mod}_CFLAGS_OTHER})
    target_include_directories(wintunes PRIVATE ${${mod}_INCLUDE_DIRS})
    target_link_directories(wintunes PRIVATE ${${mod}_LIBRARY_DIRS})
    target_link_libraries(wintunes PRIVATE ${${mod}_LIBRARIES})
endforeach()

if(WINTUNES_ENABLE_IPOD AND GPOD_FOUND AND NOT WIN32)
    target_compile_definitions(wintunes PRIVATE WINTUNES_HAVE_GPOD=1)
    target_compile_options(wintunes PRIVATE ${GPOD_CFLAGS_OTHER})
    target_include_directories(wintunes PRIVATE ${GPOD_INCLUDE_DIRS})
    target_link_directories(wintunes PRIVATE ${GPOD_LIBRARY_DIRS})
    target_link_libraries(wintunes PRIVATE ${GPOD_LIBRARIES})
else()
    target_compile_definitions(wintunes PRIVATE WINTUNES_NO_GPOD=1)
endif()
if(WINTUNES_ENABLE_CD AND CDIO_FOUND AND CDIO_PARA_FOUND AND CDDB_FOUND AND NOT WIN32)
    target_compile_definitions(wintunes PRIVATE WINTUNES_HAVE_CDIO=1)
    foreach(mod CDIO CDIO_PARA CDDB)
        target_compile_options(wintunes PRIVATE ${${mod}_CFLAGS_OTHER})
        target_include_directories(wintunes PRIVATE ${${mod}_INCLUDE_DIRS})
        target_link_directories(wintunes PRIVATE ${${mod}_LIBRARY_DIRS})
        target_link_libraries(wintunes PRIVATE ${${mod}_LIBRARIES})
    endforeach()
else()
    target_compile_definitions(wintunes PRIVATE WINTUNES_NO_CDIO=1)
endif()
if(WINTUNES_ENABLE_RADIO AND VLC_FOUND)
    target_compile_definitions(wintunes PRIVATE WINTUNES_HAVE_VLC=1)
    target_compile_options(wintunes PRIVATE ${VLC_CFLAGS_OTHER})
    target_include_directories(wintunes PRIVATE ${VLC_INCLUDE_DIRS})
    target_link_directories(wintunes PRIVATE ${VLC_LIBRARY_DIRS})
    target_link_libraries(wintunes PRIVATE ${VLC_LIBRARIES})
else()
    target_compile_definitions(wintunes PRIVATE WINTUNES_NO_VLC=1)
endif()
target_include_directories(wintunes PRIVATE src/)

if(WIN32)
    target_compile_definitions(wintunes PRIVATE _WIN32_WINNT=0x0601)
    target_link_libraries(wintunes PRIVATE shell32 ole32)
    set_target_properties(wintunes PROPERTIES WIN32_EXECUTABLE TRUE OUTPUT_NAME "WinTunes")
else()
    set_target_properties(wintunes PROPERTIES OUTPUT_NAME "wintunes")
endif()

install(TARGETS wintunes DESTINATION bin)
if(NOT WIN32)
    install(FILES resources/wintunes.desktop DESTINATION share/applications OPTIONAL)
endif()
message(STATUS "WinTunes configured (iPod=${WINTUNES_ENABLE_IPOD} / ${GPOD_FOUND}, CD=${WINTUNES_ENABLE_CD} / ${CDIO_FOUND}, Radio=${WINTUNES_ENABLE_RADIO} / ${VLC_FOUND})")
